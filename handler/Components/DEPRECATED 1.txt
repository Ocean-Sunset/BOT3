import os
import asyncio
import discord
from discord.ext import commands
import time

SIGNALS_DIR = os.path.join(os.path.dirname(__file__), "../../default/signals")
HANDLER_SIGNALS = os.path.join(os.path.dirname(__file__), "signals")

class Force(commands.Cog):
    def __init__(self, bot):
        self.bot = bot

    @commands.group(invoke_without_command=True)
    @commands.check(lambda ctx: ctx.author.id == 1234567890)  # replace with utils.is_owner
    async def force(self, ctx):
        await ctx.send("⚡ Available subcommands: `restart`, `update`, `shutdown`")

    @force.command(name="restart")
    async def force_restart(self, ctx):
        """Stylish restart of the main bot (cancelable)"""
        request_id = int(time.time())  # unique ID per request
        msg = await ctx.send("🔄 Preparing restart...")

        # Write restart signal
        signal_path = os.path.join(SIGNALS_DIR, "restart.txt")
        with open(signal_path, "w") as f:
            f.write(
                f"request_id={request_id}\n"
                f"command=restart\n"
                f"user={ctx.author}\n"
                f"timestamp={int(time.time())}\n"
                f"cancel=false\n"
            )

        spinner = ["|", "/", "-", "\\"]
        start_time = time.time()
        timeout = 30  # seconds

        while True:
            # Check for timeout
            if time.time() - start_time > timeout:
                await msg.edit(content="# ❌ ERR_444 - No response, request timed out.")
                break

            # Read bot response
            status_file = os.path.join(HANDLER_SIGNALS, "restart_status.txt")
            if os.path.exists(status_file):
                try:
                    with open(status_file, "r") as f:
                        lines = f.read().splitlines()
                except:
                    continue

                for line in lines:
                    if line.startswith("request_id=") and int(line.split("=")[1]) != request_id:
                        continue  # skip other requests

                    if line.startswith("status="):
                        status_msg = line[len("status="):]

                        # Cancel detected
                        if status_msg.lower() == "cancelled":
                            await msg.edit(content="# ❌ Restart cancelled by bot (ERR_499)")
                            os.remove(signal_path)
                            os.remove(status_file)
                            return

                        # Display spinner + progress
                        for spin in spinner:
                            await msg.edit(content=f"# {spin} {status_msg}")
                            await asyncio.sleep(0.7)

                        # If completed successfully
                        if status_msg == "OK":
                            await msg.edit(content="# ✅ Restart complete!")
                            os.remove(signal_path)
                            os.remove(status_file)
                            return

            await asyncio.sleep(1)
